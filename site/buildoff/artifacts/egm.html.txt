  <div id="status" class="sr-only" aria-live="polite" aria-atomic="true"></div>

<script>
(() => {
  const taskTitleInput = document.getElementById('task-title');
  const taskPrioritySelect = document.getElementById('task-priority');
  const addForm = document.getElementById('add-form');
  const errorDiv = document.getElementById('error');
  const taskList = document.getElementById('task-list');
  const emptyState = document.getElementById('empty-state');
  const statusDiv = document.getElementById('status');
  const filterButtons = {
    all: document.getElementById('filter-all'),
    active: document.getElementById('filter-active'),
    done: document.getElementById('filter-done')
  };
  const searchInput = document.getElementById('task-search');
  const countTotal = document.getElementById('count-total');
  const countActive = document.getElementById('count-active');
  const countDone = document.getElementById('count-done');

  let tasks = [];
  let nextId = 1;
  let currentFilter = 'all';
  let searchTerm = '';

  const priorityClass = {
    High: 'priority-high',
    Medium: 'priority-medium',
    Low: 'priority-low'
  };

  function announce(msg) {
    statusDiv.textContent = '';
    requestAnimationFrame(() => {
      statusDiv.textContent = msg;
    });
  }

  function addTask(title, priority) {
    tasks.push({id: nextId++, title, priority, completed: false});
    announce('Task added');
  }

  function deleteTask(id) {
    tasks = tasks.filter(t => t.id !== id);
    announce('Task deleted');
  }

  function toggleComplete(id, completed) {
    const task = tasks.find(t => t.id === id);
    if (task) {
      task.completed = completed;
      announce(completed ? 'Task completed' : 'Task marked as active');
    }
  }

  function updateCounters() {
    countTotal.textContent = tasks.length;
    countActive.textContent = tasks.filter(t => !t.completed).length;
    countDone.textContent = tasks.filter(t => t.completed).length;
  }

  function render() {
    const term = searchTerm.trim().toLowerCase();
    const filtered = tasks.filter(t => {
      const matchesSearch = !term || t.title.toLowerCase().includes(term);
      const matchesFilter = currentFilter === 'all' ||
        (currentFilter === 'active' && !t.completed) ||
        (currentFilter === 'done' && t.completed);
      return matchesSearch && matchesFilter;
    });

    taskList.innerHTML = '';
    emptyState.hidden = filtered.length !== 0;

    filtered.forEach(task => {
      const li = document.createElement('li');
      li.className = 'task-item' + (task.completed ? ' completed' : '');
      li.dataset.taskId = task.id;

      const infoDiv = document.createElement('div');
      infoDiv.className = 'task-info';

      const checkbox = document.createElement('input');
      checkbox.type = 'checkbox';
      checkbox.checked = task.completed;
      checkbox.id = `chk-${task.id}`;
      checkbox.addEventListener('change', () => {
        toggleComplete(task.id, checkbox.checked);
        render();
        updateCounters();
      });

      const titleSpan = document.createElement('span');
      titleSpan.className = 'title';
      titleSpan.textContent = task.title;

      const prioritySpan = document.createElement('span');
      prioritySpan.className = `priority ${priorityClass[task.priority] || ''}`;
      prioritySpan.textContent = task.priority;

      const label = document.createElement('label');
      label.htmlFor = checkbox.id;
      label.append(checkbox, titleSpan);

      infoDiv.append(label, prioritySpan);

      const delBtn = document.createElement('button');
      delBtn.type = 'button';
      delBtn.setAttribute('aria-label', 'Delete task');
      delBtn.textContent = 'Delete';
      delBtn.addEventListener('click', () => {
        deleteTask(task.id);
        render();
        updateCounters();
      });

      li.append(infoDiv, delBtn);
      taskList.appendChild(li);
    });

    updateCounters();
  }

  // Initial tasks
  addTask('Outline the launch plan', 'High');
  addTask('Take a walking break', 'Low');
  addTask('Review the first draft', 'Medium');
  render();

  // Form handling
  addForm.addEventListener('submit', e => {
    e.preventDefault();
    const rawTitle = taskTitleInput.value;
    const title = rawTitle.trim();
    if (!title) {
      errorDiv.textContent = 'Task title cannot be empty.';
      taskTitleInput.focus();
      return;
    }
    errorDiv.textContent = '';
    const priority = taskPrioritySelect.value;
    addTask(title, priority);
    taskTitleInput.value = '';
    taskPrioritySelect.value = 'Medium';
    render();
    updateCounters();
  });

  // Filter buttons
  Object.entries(filterButtons).forEach(([key, btn]) => {
    btn.addEventListener('click', () => {
      currentFilter = key;
      Object.values(filterButtons).forEach(b => b.setAttribute('aria-pressed', 'false'));
      btn.setAttribute('aria-pressed', 'true');
      render();
    });
  });

  // Search handling
  searchInput.addEventListener('input', () => {
    searchTerm = searchInput.value;
    render();
  });

  // Clear error on input
  taskTitleInput.addEventListener('input', () => {
    if (errorDiv.textContent) errorDiv.textContent = '';
  });
})();
</script>
